14642:     def _deep_search(source: Any, keys: List[str]) -> Optional[str]:
14643:         if source is None:
14644:             return None
14645:         key_set = {key.casefold() for key in keys}
14646:         stack: List[Any] = [source]
14647:         seen: Set[int] = set()
14648:         while stack:
14649:             current = stack.pop()
14650:             ident = id(current)
14651:             if ident in seen:
14652:                 continue
14653:             seen.add(ident)
14654:             if isinstance(current, dict):
14655:                 for key, val in current.items():
14656:                     key_cf = key.casefold() if isinstance(key, str) else None
14657:                     if key_cf and key_cf in key_set:
14658:                         candidate = _extract_text(val)
14659:                         if candidate:
14660:                             return candidate
14661:                     stack.append(val)
14662:             elif isinstance(current, (list, tuple)):
14663:                 stack.extend(current)
14664:         return None
