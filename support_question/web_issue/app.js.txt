// --- Auth & Session ---
async function ensureSession() {
  let sid = localStorage.getItem(SESSION_KEY);
  if (!sid) sid = await ensureLoginSession();
  return sid;
}

async function authFetch(url, opts = {}) {
  const sid = await ensureSession();
  const headers = { ...(opts.headers || {}), "X-Session-Id": sid };
  return fetch(url, { ...opts, headers });
}

async function fetchSessionInfo() {
  try {
    const res = await authFetch("/web/api/session");
    if (res.status === 401) { localStorage.removeItem(SESSION_KEY); location.reload(); return; }
    if (res.ok) {
        const info = await res.json();
        localStorage.setItem(SESSION_INFO_KEY, JSON.stringify(info));
        applySessionInfo(info);
    }
  } catch (e) { console.warn("Session check failed", e); }
}

function applySessionInfo(info) {
  if (!info) return;
  isAdmin = !!info.is_admin;
  if (userLabelEl) userLabelEl.textContent = info.display_name || info.login;
  // Show admin buttons
  if (adminBtn) adminBtn.style.display = isAdmin ? "inline-block" : "none";
  if (shadowToggleBtn) shadowToggleBtn.style.display = isAdmin ? "inline-block" : "none";
}

// --- Data Loading ---
async function loadUnits({ restartLive = false } = {}) {
  const res = await authFetch("/web/api/units");
  if (!res.ok) throw new Error("API Error " + res.status);
  units = await res.json();
  // Cache previews
  cardPreviewById = new Map();
  units.forEach(u => {
      if(u.card_preview) { cardPreviewById.set(u.id, u.card_preview); delete u.card_preview; }
  });
  
  unitsById.clear();
  units.forEach(u => unitsById.set(u.id, u));
  
  // Initial sort
  units.sort((a, b) => a.name.localeCompare(b.name, "ru", { sensitivity: "base" }));
  
  applyFilters();
  lastFeedTick = Date.now();
  
  if (restartLive) {
     feedSince = null;
     scheduleLiveUpdates(true);
  }
}

function scheduleLiveUpdates(reset = false) {
  if (reset && liveTimer) { clearTimeout(liveTimer); liveTimer = null; }
  if (liveTimer) return;
  liveTimer = setTimeout(fetchLiveUpdates, FEED_INTERVAL_MS);
}

async function fetchLiveUpdates() {
  if (liveInFlight) { scheduleLiveUpdates(); return; }
  liveInFlight = true;
  try {
    const watchIds = getWatchIds().slice(0, 400);
    const qs = [];
    if (feedSince) qs.push(`since=${feedSince}`);
    if (watchIds.length) qs.push(`watch_ids=${watchIds.join(",")}`);
    const url = `/web/api/units/feed${qs.length ? "?"+qs.join("&") : ""}`;
    const res = await authFetch(url);
    if (res.ok) {
        const data = await res.json();
        lastFeedTick = Date.now();
        if (typeof data.shadow_count === "number") updateShadowBadge(data.shadow_count);
        if (data.ts) feedSince = data.ts;
        if (data.reset) await loadUnits({ restartLive: true });
        else if (data.updates?.length) applyFeedUpdates(data.updates);
    }
  } catch (e) { console.warn("Feed error", e); }
  finally {
    liveInFlight = false;
    liveTimer = null;
    scheduleLiveUpdates();
  }
}

function applyFeedUpdates(updates) {
  let changed = false;
  const touched = new Set();
  updates.forEach(patch => {
    const u = unitsById.get(patch.id);
    if (!u) return;
    touched.add(patch.id);
    // Merge fields
    for (const k of ["online", "status", "status_label", "last_ts", "speed", "lat", "lon", "ignition", "has_fuel"]) {
        if (patch[k] !== undefined && u[k] !== patch[k]) { u[k] = patch[k]; changed = true; }
    }
  });
  if (changed) {
      applyFilters();
      if (selectedId && touched.has(selectedId)) refreshCurrentCard();
  }
}

function renderCard(detail) {
  if (!cardEl) return;
  const item = detail.item;
  const status = detail.status_details || {};
  
  // Prepare data
  const statusTxt = status.status_label || "Нет связи";
  const lastTime = formatRelativeTime(status.last_ts);
  const statusCls = status.online ? "online" : "offline";

  const sections = [];
  
  // Location
  const addr = detail.card_data?.location?.address || "—";
  const coords = item.lat ? `${item.lat.toFixed(5)}, ${item.lon.toFixed(5)}` : "—";
  sections.push({
      key: "location", title: "Местоположение",
      html: renderKvGrid([{label: "Адрес", value: addr}, {label: "Координаты", value: coords}])
  });

  // Sensors
  if(cardSettings.sensors) {
      const sens = detail.card_data?.sensors || [];
      if(sens.length) sections.push({
          key: "sensors", title: "Датчики",
          html: renderKvGrid(sens.map(s => ({label: s.name, value: s.value + (s.units ? " "+s.units : "")})))
      });
  }

  // Params (filtered)
  if(cardSettings.params) {
      const prms = (detail.card_data?.params || []).slice(0, 20);
      if(prms.length) sections.push({
          key: "params", title: "Параметры",
          html: renderKvGrid(prms.map(p => ({label: p.key, value: p.value})))
      });
  }
  
  // Render
  const actionsHtml = `
     <div class="card-actions">
        <button class="ghost icon" data-action="open-history" title="История">📅</button>
        <button class="ghost icon" data-action="card-layout-toggle" title="Настройки">⚙️</button>
        <button class="ghost icon" data-action="card-close" title="Закрыть">✕</button>
     </div>
  `;
  
  const sectionsHtml = sections.map(s => `
    <div class="card-block" data-key="${s.key}">
       <div class="label">${s.title}</div>
       <div class="card-block-body">${s.html}</div>
    </div>
  `).join("");

  cardEl.innerHTML = `
    <div class="card-head">
        <div class="card-head-main">
            <div class="card-head-name">${item.name}</div>
            <div class="card-head-subtitle">#${item.id} ${item.hw||""}</div>
        </div>
        ${actionsHtml}
    </div>
    <div class="card-status-row">
        <span class="card-status-dot ${statusCls}"></span>
        <div class="card-status-body">
            <div class="card-status-title">${statusTxt}</div>
            <div class="card-status-caption">${lastTime}</div>
        </div>
    </div>
    <div class="card-grid">${sectionsHtml}</div>
  `;
  
  cardEl.classList.remove("hidden");
  selectedId = item.id;
  currentCardDetail = detail;
  
  // Position card relative to row
  const row = listEl.querySelector(`.unit-row[data-id="${item.id}"]`);
  if(row) {
      const top = row.offsetTop + ROW_H + CARD_ANCHOR_OFFSET;
      cardEl.style.top = top + "px";
      cardVisible = true;
      expandedIndex = parseInt(row.dataset.index);
      innerEl.style.height = (current.length * ROW_H + cardEl.offsetHeight + 20) + "px";
      
      // Shift rows below
      Array.from(innerEl.children).forEach(el => {
          if(el.classList.contains("unit-row")) {
             const idx = parseInt(el.dataset.index);
             let y = idx * ROW_H;
             if(idx > expandedIndex) y += (cardEl.offsetHeight + CARD_GAP);
             el.style.transform = `translateY(${y}px)`;
          }
      });
  }
}

function ensureListDom() {

function applyFilters() {
    const q = document.getElementById("search").value.toLowerCase();
    current = units.filter(u => !q || u.name.toLowerCase().includes(q));
    
    renderList();
    syncMarkers(current);
    
    const sEl = document.getElementById("status");
    if(sEl) sEl.textContent = `Всего: ${units.length} / Отображено: ${current.length}`;
}

function renderList() {
    if(!listEl) return;
    const viewportH = listEl.clientHeight;
    const scrollY = listEl.scrollTop;
    
    // Basic virtualization
    const startIdx = Math.max(0, Math.floor(scrollY / ROW_H) - OVERSCAN);
    const endIdx = Math.min(current.length, Math.ceil((scrollY + viewportH) / ROW_H) + OVERSCAN);
    
    const cardH = cardVisible ? (cardEl.offsetHeight + CARD_GAP) : 0;
    innerEl.style.height = (current.length * ROW_H + cardH) + "px";
    
    // Clear pool usage
    pool.forEach(el => el.style.display = 'none');
    
    // Ensure pool size
    while(pool.length < (endIdx - startIdx)) {
        const el = document.createElement("div");
        el.className = "unit-row";
        el.innerHTML = `<div class="unit-name"></div><div class="unit-meta"></div><div class="unit-meta" style="text-align:right"></div>`;
        el.addEventListener("click", onRowClick);
        innerEl.appendChild(el);
        pool.push(el);
    }
    
    let poolIdx = 0;
    for(let i=startIdx; i<endIdx; i++) {
        const u = current[i];
        const el = pool[poolIdx++];
        el.style.display = 'grid';
        el.dataset.id = u.id;
        el.dataset.index = i;
        
        let y = i * ROW_H;
        if(cardVisible && expandedIndex !== null && i > expandedIndex) y += cardH;
        el.style.transform = `translateY(${y}px)`;
        el.classList.toggle("is-selected", u.id === selectedId);
        
        // Fill data
        const nameEl = el.children[0];
        nameEl.textContent = u.name;
        if(u.online) nameEl.style.color = "#fff"; else nameEl.style.color = "#aaa";
        
        const metaEl = el.children[1];
        metaEl.innerHTML = `<span class="dot ${u.online?'online':'offline'}"></span> ${u.status_label || "—"}`;
        
        const speedEl = el.children[2];
        speedEl.textContent = u.speed > 0 ? Math.round(u.speed)+" км/ч" : "";
    }
}

function onRowClick(e) {
    const id = parseInt(e.currentTarget.dataset.id);
    selectUnit(id);
}

// --- Shadow / Inbox ---
async function toggleShadowMode() {
    if(mode === "monitor") {
        mode = "inbox";
        hideCard();
        await loadShadowInbox();
        renderShadowSidebar();
    } else {
        mode = "monitor";
        await loadUnits();
    }
}

async function loadShadowInbox() {
    try {
        const res = await authFetch("/web/api/unknown_devices");
        shadowUnits = await res.json();
        updateShadowBadge(shadowUnits.length);
    } catch (e) { console.error(e); }
}

function renderShadowSidebar() {
    // Reuse list rendering logic or simpler substitution
    innerEl.innerHTML = ""; // Clear virtualization
    if (!shadowUnits.length) {
        innerEl.innerHTML = `<div style="padding:20px;text-align:center;color:#888">Нет новых устройств</div>`;
        return;
    }
    shadowUnits.forEach(u => {
        const el = document.createElement("div");
        el.className = "shadow-row"; // CSS class needed
        el.innerHTML = `
            <div class="shadow-uid">${u.uid}</div>
            <div class="shadow-proto">${u.protocol}</div>
            <button class="primary small" onclick="app.shadowCreate('${u.protocol}', '${u.uid}')">Создать</button>
        `;
        innerEl.appendChild(el);
    });
}

async function loadHistoryTrips() {
    if (!histUnitId) return;
    
    const listEl = document.getElementById("history-timeline");
    if(listEl) listEl.innerHTML = '<div class="muted" style="padding:20px; text-align:center">Загрузка...</div>';
    const sumEl = document.getElementById("history-summary");
    if(sumEl) sumEl.innerHTML = '...';
    
    clearHistoryMap();
    trackPointsCache = [];

    // Начало и конец дня
    const from = new Date(histDate); from.setHours(0,0,0,0);
    const to = new Date(histDate); to.setHours(23,59,59,999);
    
    const fromTs = Math.floor(from.getTime()/1000);
    const toTs = Math.floor(to.getTime()/1000);

    try {
        const res = await authFetch(`/web/api/history/trips?unit_id=${histUnitId}&from_ts=${fromTs}&to_ts=${toTs}`);
        if (!res.ok) throw new Error("HTTP Error");
        const trips = await res.json();
        
        renderTimeline(trips);
        calculateDailyStats(trips);
        
    } catch (e) {
        console.error(e);
        if(listEl) listEl.innerHTML = '<div class="muted" style="text-align:center; padding:20px; color:#ef5350">Ошибка загрузки</div>';
    }
}

function calculateDailyStats(trips) {
    let dist = 0;
    let maxSpd = 0;
    trips.forEach(t => {
        if(t.type === 'trip') {
            dist += (t.distance_m || 0);
            if(t.max_speed > maxSpd) maxSpd = t.max_speed;
        }
    });
    const distKm = (dist / 1000).toFixed(1);
    const sumEl = document.getElementById("history-summary");
    if(sumEl) sumEl.innerHTML = `
        <span class="sum-item"><span class="sum-val">${distKm}</span> <span class="sum-lbl">Пробег (км)</span></span>
        <span class="sum-item"><span class="sum-val">${maxSpd}</span> <span class="sum-lbl">Макс (км/ч)</span></span>
    `;
}

function renderTimeline(trips) {
    const listEl = document.getElementById("history-timeline");
    if(!listEl) return;
    listEl.innerHTML = "";
    
    if (!trips || !trips.length) {
        listEl.innerHTML = '<div class="muted" style="padding:20px; text-align:center">Нет поездок за этот день</div>';
        return;
    }

    trips.forEach(t => {
        const el = document.createElement("div");
        const isTrip = t.type === 'trip';
        el.className = `timeline-item ${isTrip ? 'trip' : (t.type === 'stop' ? 'stop' : 'stay')}`;
        
        const start = new Date(t.start_ts * 1000).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
        const end = new Date(t.end_ts * 1000).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
        
        let detailHtml = "";
        let icon = "";
        
        if (isTrip) {
            icon = "🚗";
            const km = (t.distance_m / 1000).toFixed(1);
            detailHtml = `${km} км &bull; макс ${t.max_speed} км/ч`;
        } else {
            icon = t.type === 'stop' ? "🛑" : "🅿️";
            detailHtml = t.type === 'stop' ? 'Остановка' : 'Стоянка';
        }
        
        const addr = t.start_address || (isTrip ? "Начало движения" : "Адрес определяется...");

        el.innerHTML = `
            <div class="t-icon">${icon}</div>
            <div class="t-row">
                <span class="t-time">${start} - ${end}</span>
                <span class="t-dur">${t.duration_str}</span>
            </div>
            <div class="t-detail">${detailHtml}</div>
            <div class="t-addr">${addr}</div>
        `;
        
        el.addEventListener("click", () => {
            listEl.querySelectorAll(".timeline-item").forEach(x => x.classList.remove("active"));
            el.classList.add("active");
            loadTrackOnMap(t);
        });
        
        listEl.appendChild(el);
    });
}

async function loadTrackOnMap(trip) {
    clearHistoryMap();
    if (!map) return;
    
    // Скрываем инфо-плашку пока грузим
    document.getElementById("track-cursor-info")?.classList.add("hidden");

    try {
        const res = await authFetch(`/web/api/history/track?unit_id=${trip.unit_id}&from_ts=${trip.start_ts}&to_ts=${trip.end_ts}`);
        const points = await res.json();
        
        if (!points.length) {
            showToast("Нет точек трека");
            return;
        }
        
        // Сохраняем для интерактива
        trackPointsCache = points;

        histLayerGroup = L.layerGroup().addTo(map);

        // 1. Линия трека
        const latlngs = points.map(p => [p.lat, p.lon]);
        L.polyline(latlngs, {
            color: '#3b82f6', 
            weight: 4, 
            opacity: 0.8,
            lineJoin: 'round'
        }).addTo(histLayerGroup);

        // 2. Маркеры Старт (A) / Финиш (B)
        const startPt = latlngs[0];
        const endPt = latlngs[latlngs.length - 1];

        createMarker(startPt, "A", "#4caf50").addTo(histLayerGroup); // Зеленый
        createMarker(endPt, "B", "#f44336").addTo(histLayerGroup);   // Красный

        // 3. Стрелки (исправленный расчет через проекцию экрана)
        drawArrows(points, histLayerGroup);

        // Зум к треку
        map.fitBounds(L.latLngBounds(latlngs), { padding: [50, 50] });
        
        showToast("Трек загружен");

    } catch(e) {
        console.error(e);
        showToast("Ошибка загрузки трека");
    }
}

function createMarker(latlng, label, color) {
    return L.marker(latlng, {
        icon: L.divIcon({
            className: 'map-label-icon',
            html: `<div style="background:${color}; width:24px; height:24px; border-radius:50%; border:2px solid #fff; color:#fff; font-weight:bold; display:flex; align-items:center; justify-content:center; box-shadow:0 2px 5px rgba(0,0,0,0.5); font-size:12px;">${label}</div>`,
            iconSize: [24, 24],
            iconAnchor: [12, 12]
        })
    });
}

function drawArrows(points, layerGroup) {
    if (points.length < 2) return;
    
    // Рисуем стрелку каждые ~20 точек или 500-1000 метров, чтобы не спамить
    const step = Math.max(10, Math.floor(points.length / 15)); 
    
    for (let i = 0; i < points.length - 1; i += step) {
        const p1 = points[i];
        const p2 = points[i+1];
        
        // Считаем угол через экранные координаты Leaflet для точности отображения
        const pp1 = map.latLngToLayerPoint([p1.lat, p1.lon]);
        const pp2 = map.latLngToLayerPoint([p2.lat, p2.lon]);
        
        const dx = pp2.x - pp1.x;
        const dy = pp2.y - pp1.y;
        
        // Если точки слишком близко на экране, стрелка будет кривой, пропускаем
        if (dx*dx + dy*dy < 20) continue; 

        // Угол в градусах (0 = вправо, 90 = вниз)
        const angle = Math.atan2(dy, dx) * 180 / Math.PI;
        
        L.marker([p2.lat, p2.lon], {
            icon: L.divIcon({
                className: '',
                html: `<div class="arrow-icon" style="transform: rotate(${angle}deg);">➤</div>`,
                iconSize: [20, 20],
                iconAnchor: [10, 10]
            }),
            interactive: false // Чтобы не мешали наведению
        }).addTo(layerGroup);
    }
}

function clearHistoryMap() {
    if (histLayerGroup) {
        map.removeLayer(histLayerGroup);
        histLayerGroup = null;
    }
    if (trackCursorMarker) {
        map.removeLayer(trackCursorMarker);
        trackCursorMarker = null;
    }
    document.getElementById("track-cursor-info")?.classList.add("hidden");
}

// Обработчик движения мыши по карте для интерактива трека
function onMapMouseMove(e) {
    if (!trackPointsCache.length || !histLayerGroup) return;
    
    // Ищем ближайшую точку трека к курсору мыши
    let minDist = Infinity;
    let closest = null;
    const mousePoint = map.latLngToLayerPoint(e.latlng);
    
    // Оптимизация: бежим не по всем точкам, если их очень много, но 5000 JS переварит
    for (const pt of trackPointsCache) {
        const layerPoint = map.latLngToLayerPoint([pt.lat, pt.lon]);
        const dx = layerPoint.x - mousePoint.x;
        const dy = layerPoint.y - mousePoint.y;
        const distSq = dx*dx + dy*dy;
        
        if (distSq < minDist) {
            minDist = distSq;
            closest = pt;
        }
    }
    
    // Если курсор ближе 20 пикселей к треку
    if (closest && minDist < 400) { 
        showTrackCursor(closest);
    } else {
        hideTrackCursor();
    }
}

function showTrackCursor(pt) {
    const infoBox = document.getElementById("track-cursor-info");
    if (!infoBox) return;
    
    // Обновляем текст
    const timeStr = new Date(pt.ts * 1000).toLocaleTimeString();
    infoBox.querySelector(".ti-time").textContent = timeStr;
    infoBox.querySelector(".ti-speed").textContent = Math.round(pt.speed || 0) + " км/ч";
    
    infoBox.classList.remove("hidden");
    
    // Рисуем/двигаем кружок на карте
    if (!trackCursorMarker) {
        trackCursorMarker = L.circleMarker([pt.lat, pt.lon], {
            radius: 6,
            color: '#fff',
            fillColor: '#4caf50',
            fillOpacity: 1,
            weight: 2,
            interactive: false
        }).addTo(map);
    } else {
        trackCursorMarker.setLatLng([pt.lat, pt.lon]);
        trackCursorMarker.bringToFront();
    }
}

function hideTrackCursor() {
    document.getElementById("track-cursor-info")?.classList.add("hidden");
    if (trackCursorMarker) {
        map.removeLayer(trackCursorMarker);
        trackCursorMarker = null;
    }
}

// --- Auth / Login UI ---
let loginInitDone = false;
let loginOverlayEl, loginErrorEl, loginLoginInput, loginPasswordInput, loginSubmitBtn;

function initLoginModal() {
  if (loginInitDone) return;
  loginInitDone = true;
  loginOverlayEl = document.getElementById("login-overlay");
  loginErrorEl = document.getElementById("login-error");
  loginLoginInput = document.getElementById("login-login");
  loginPasswordInput = document.getElementById("login-password");
  loginSubmitBtn = document.getElementById("login-submit");
  if (!loginOverlayEl) return;
  
  loginSubmitBtn.addEventListener("click", doLogin);
  document.getElementById("login-form").addEventListener("submit", e => {
      e.preventDefault(); doLogin();
  });
}

async function doLogin() {
    const login = loginLoginInput.value.trim();
    const password = loginPasswordInput.value;
    if (!login || !password) { loginErrorEl.textContent = "Введите логин и пароль"; return; }
    
    loginErrorEl.textContent = "";
    loginSubmitBtn.disabled = true;
    try {
        const res = await fetch("/web/api/login", {
            method: "POST", headers: {"Content-Type": "application/json"},
            body: JSON.stringify({login, password})
        });
        if (res.ok) {
            const data = await res.json();
            localStorage.setItem(SESSION_KEY, data.session_id);
            localStorage.setItem(SESSION_INFO_KEY, JSON.stringify(data));
            loginOverlayEl.classList.add("hidden");
            applySessionInfo(data);
        } else {
            loginErrorEl.textContent = "Ошибка входа";
        }
    } catch(e) { loginErrorEl.textContent = "Ошибка сети"; }
    finally { loginSubmitBtn.disabled = false; }
}

function showLoginModal() {
    initLoginModal();
    loginOverlayEl?.classList.remove("hidden");
}

async function ensureLoginSession() {
    return new Promise(resolve => {
        const sid = localStorage.getItem(SESSION_KEY);
        if(sid) resolve(sid);
        else showLoginModal();
        // Poll for session
        const i = setInterval(() => {
            const s = localStorage.getItem(SESSION_KEY);
            if(s) { clearInterval(i); resolve(s); }
        }, 500);
    });
}

function formatRelativeTime(ts) {
    if(!ts) return "";
    const diff = Math.floor(Date.now()/1000 - ts);
    if(diff < 60) return `${diff} сек назад`;
    if(diff < 3600) return `${Math.floor(diff/60)} мин назад`;
    return `${Math.floor(diff/3600)} ч назад`;
}
